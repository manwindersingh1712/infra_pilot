# Plan 2 — Go API, Auth Hardening and UI

> **For agentic workers:** REQUIRED: execute this plan with the team orchestration in spec §13 (`docs/superpowers/specs/2026-10-07-go-temporal-design.md`). The Engineer follows `docs/superpowers/agents/engineer.md`. After every task, Security, Staff and Verifier review in parallel. **Product** also reviews every task that changes API responses or the UI, which is most of this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Node API with the Go API, and close the security gaps from spec §5: token expiry with refresh tokens, ownership on every route, encrypted and masked env vars, one-time SSE tickets, rate limiting, a CORS allow-list and dev-login gating. Add service deletion, and move the UI to SSE and the new auth flow.

**Architecture:** `internal/api` serves the same routes the UI uses today, plus `POST /auth/refresh`, `POST /auth/logout`, `DELETE /services/:id` and `POST /logs/deployments/:id/stream-ticket`. All writes that need background work go through the outbox, which Plan 1's worker turns into Temporal workflows. This plan adds `TeardownWorkflow` for deletes. Redis holds stream tickets. The UI gets one shared API client that refreshes tokens automatically.

**Tech Stack:** Go (net/http routing patterns), `golang-jwt/jwt/v5`, `golang.org/x/crypto/bcrypt`, `golang.org/x/time/rate`, `redis/go-redis/v9`, `prometheus/client_golang`, `go.temporal.io/sdk`. UI: React 19 + Vite with the browser's `EventSource`.

**Prerequisite:** Plan 1 is complete and its worker cutover is done. The Go worker is running and the rollback window is open. The Node API keeps serving the UI until this plan's cutover (Task 18).

## Background for the implementer

- Read spec §5 (security), §7 (data model, migration `00003`) and §9 (rollout).
- The cutover at the end of this plan is the **point of no return for Node**. From then on, the UI stores a refresh token, tokens without `exp` are rejected, and env values are encrypted.
- **API contract.** The UI (`cp-ui/src`) relies on these response shapes. Keep them unless a task changes the UI in the same commit:
  - Prisma-style JSON with camelCase keys and `toISOString()` timestamps;
  - `[]` for empty lists;
  - errors shaped `{"error": "..."}`, with the error strings shown in each task.
- **Ownership.** Every route except `/healthz`, `/readyz`, `/auth/*`, `/dev/login` (development only) and the ticket-authenticated stream checks that the caller owns the resource. A foreign or missing ID gets the same 404, so callers cannot probe which IDs exist.
- The test harness (Task 5) keeps the helper names from the old plan (`newHarness`, `hs.token`, `hs.do`, `decode`, `expectStatus`, `expectBody`), so reused tests run unchanged.
- **Conventions:**
  - `go` commands run in `control-plane-go/`.
  - `npm` commands run in `cp-ui/`.
  - Always test with `-race`.
  - Every task ends with a **Review gate** step.

```go
package store

import (
	"context"

	"github.com/jackc/pgx/v5"
	"github.com/lucsky/cuid"
)

const connCols = `"id", "sourceId", "targetId", "label", "createdAt"`

func scanConnection(row pgx.Row) (ServiceConnection, error) {
	var c ServiceConnection
	err := row.Scan(&c.ID, &c.SourceID, &c.TargetID, &c.Label, &c.CreatedAt.Time)
	return c, noRows(err)
}

// CreateConnection returns ErrConflict when the pair already exists.
func (s *Store) CreateConnection(ctx context.Context, sourceID, targetID string, label *string) (ServiceConnection, error) {
	c, err := scanConnection(s.db.QueryRow(ctx,
		`INSERT INTO "ServiceConnection" ("id", "sourceId", "targetId", "label") VALUES ($1, $2, $3, $4)
		 RETURNING `+connCols, cuid.New(), sourceID, targetID, label))
	if isUniqueViolation(err) {
		return ServiceConnection{}, ErrConflict
	}
	return c, err
}

// ListProjectConnections returns connections touching any service of the project.
func (s *Store) ListProjectConnections(ctx context.Context, projectID string) ([]ServiceConnection, error) {
	rows, err := s.db.Query(ctx,
		`SELECT `+connCols+` FROM "ServiceConnection"
		 WHERE "sourceId" IN (SELECT "id" FROM "Service" WHERE "projectId" = $1)
		    OR "targetId" IN (SELECT "id" FROM "Service" WHERE "projectId" = $1)
		 ORDER BY "createdAt" ASC`, projectID)
	return collect(rows, err, scanConnection)
}

func (s *Store) GetConnectionForOwner(ctx context.Context, id, ownerUserID string) (ServiceConnection, error) {
	return scanConnection(s.db.QueryRow(ctx,
		`SELECT `+connCols+` FROM "ServiceConnection"
		 WHERE "id" = $1 AND "sourceId" IN (
		   SELECT s."id" FROM "Service" s JOIN "Project" p ON p."id" = s."projectId" WHERE p."ownerUserId" = $2)`,
		id, ownerUserID))
}

func (s *Store) DeleteConnection(ctx context.Context, id string) error {
	_, err := s.db.Exec(ctx, `DELETE FROM "ServiceConnection" WHERE "id" = $1`, id)
	return err
}

// ListSourceEdges returns [source, target] pairs whose source is in the project.
func (s *Store) ListSourceEdges(ctx context.Context, projectID string) ([][2]string, error) {
	rows, err := s.db.Query(ctx,
		`SELECT "sourceId", "targetId" FROM "ServiceConnection"
		 WHERE "sourceId" IN (SELECT "id" FROM "Service" WHERE "projectId" = $1)
		 ORDER BY "createdAt" ASC`, projectID)
	return collect(rows, err, func(row pgx.Row) ([2]string, error) {
		var e [2]string
		err := row.Scan(&e[0], &e[1])
		return e, err
	})
}
```

```go
package store

import (
	"context"

	"github.com/jackc/pgx/v5"
	"github.com/lucsky/cuid"
)

// Every query below hides soft-deleted services ("deletedAt" IS NOT NULL).
const serviceCols = `"id", "projectId", "name", "serviceType"::text, "repoUrl", "branch", "createdAt", "positionX", "positionY"`

func scanService(row pgx.Row) (Service, error) {
	var v Service
	err := row.Scan(&v.ID, &v.ProjectID, &v.Name, &v.ServiceType, &v.RepoURL, &v.Branch,
		&v.CreatedAt.Time, &v.PositionX, &v.PositionY)
	return v, noRows(err)
}

type NewService struct {
	ProjectID   string
	Name        string
	ServiceType string
	RepoURL     *string
	Branch      string
}

func (s *Store) CreateService(ctx context.Context, n NewService) (Service, error) {
	return scanService(s.db.QueryRow(ctx,
		`INSERT INTO "Service" ("id", "projectId", "name", "serviceType", "repoUrl", "branch")
		 VALUES ($1, $2, $3, $4::"ServiceType", $5, $6) RETURNING `+serviceCols,
		cuid.New(), n.ProjectID, n.Name, n.ServiceType, n.RepoURL, n.Branch))
}

func (s *Store) GetService(ctx context.Context, id string) (Service, error) {
	return scanService(s.db.QueryRow(ctx,
		`SELECT `+serviceCols+` FROM "Service" WHERE "id" = $1 AND "deletedAt" IS NULL`, id))
}

// ListServices returns all services, or one project's when projectID is not
// empty, newest first. API handlers use ListServicesForOwner instead.
func (s *Store) ListServices(ctx context.Context, projectID string) ([]Service, error) {
	rows, err := s.db.Query(ctx,
		`SELECT `+serviceCols+` FROM "Service"
		 WHERE "deletedAt" IS NULL AND ($1 = '' OR "projectId" = $1) ORDER BY "createdAt" DESC`, projectID)
	return collect(rows, err, scanService)
}

// ListServicesForOwner returns the caller's services, optionally for one project.
func (s *Store) ListServicesForOwner(ctx context.Context, ownerUserID, projectID string) ([]Service, error) {
	rows, err := s.db.Query(ctx,
		`SELECT `+serviceCols+` FROM "Service"
		 WHERE "deletedAt" IS NULL AND ($2 = '' OR "projectId" = $2)
		   AND "projectId" IN (SELECT "id" FROM "Project" WHERE "ownerUserId" = $1)
		 ORDER BY "createdAt" DESC`, ownerUserID, projectID)
	return collect(rows, err, scanService)
}

func (s *Store) GetServiceForOwner(ctx context.Context, id, ownerUserID string) (Service, error) {
	return scanService(s.db.QueryRow(ctx,
		`SELECT `+serviceCols+` FROM "Service"
		 WHERE "id" = $1 AND "deletedAt" IS NULL
		   AND "projectId" IN (SELECT "id" FROM "Project" WHERE "ownerUserId" = $2)`,
		id, ownerUserID))
}

// SoftDeleteService hides the service; TeardownWorkflow removes its containers.
func (s *Store) SoftDeleteService(ctx context.Context, id string) error {
	tag, err := s.db.Exec(ctx,
		`UPDATE "Service" SET "deletedAt" = now() WHERE "id" = $1 AND "deletedAt" IS NULL`, id)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrNotFound
	}
	return nil
}

func (s *Store) UpdateServicePosition(ctx context.Context, id string, x, y float64) error {
	tag, err := s.db.Exec(ctx,
		`UPDATE "Service" SET "positionX" = $2, "positionY" = $3 WHERE "id" = $1 AND "deletedAt" IS NULL`, id, x, y)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrNotFound
	}
	return nil
```

```go
// ErrTokenReused means a refresh token was presented after it was consumed:
// someone else may hold a copy, so all of the user's tokens were revoked.
var ErrTokenReused = errors.New("refresh token reused")

// CreateRefreshToken stores the SHA-256 hash of a refresh token. Plain tokens are never stored.
func (s *Store) CreateRefreshToken(ctx context.Context, userID, tokenHash string, ttl time.Duration) error {
	_, err := s.db.Exec(ctx,
		`INSERT INTO "RefreshToken" ("id", "userId", "tokenHash", "expiresAt")
		 VALUES ($1, $2, $3, now() + $4::int * interval '1 second')`,
		cuid.New(), userID, tokenHash, int(ttl.Seconds()))
	return err
}

// ConsumeRefreshToken revokes a valid token and returns its user (rotation).
// A token that was already used revokes every token of its user.
func (s *Store) ConsumeRefreshToken(ctx context.Context, tokenHash string) (string, error) {
	var userID string
	err := s.db.QueryRow(ctx,
		`UPDATE "RefreshToken" SET "revokedAt" = now()
		 WHERE "tokenHash" = $1 AND "revokedAt" IS NULL AND "expiresAt" > now()
		 RETURNING "userId"`, tokenHash).Scan(&userID)
	if err == nil {
		return userID, nil
	}
	if !errors.Is(err, pgx.ErrNoRows) {
		return "", err
	}
	var revoked bool
	err = s.db.QueryRow(ctx,
		`SELECT "userId", "revokedAt" IS NOT NULL FROM "RefreshToken" WHERE "tokenHash" = $1`, tokenHash).
		Scan(&userID, &revoked)
	if errors.Is(err, pgx.ErrNoRows) || (err == nil && !revoked) {
		return "", ErrNotFound // unknown or expired
	}
	if err != nil {
		return "", err
	}
	if _, err := s.db.Exec(ctx,
		`UPDATE "RefreshToken" SET "revokedAt" = now() WHERE "userId" = $1 AND "revokedAt" IS NULL`, userID); err != nil {
		return "", err
	}
	return "", ErrTokenReused
}

func (s *Store) RevokeRefreshToken(ctx context.Context, tokenHash string) error {
	_, err := s.db.Exec(ctx,
		`UPDATE "RefreshToken" SET "revokedAt" = now() WHERE "tokenHash" = $1 AND "revokedAt" IS NULL`, tokenHash)
	return err
}
```

```go
// Package tickets issues short-lived, single-use tickets stored in Redis.
package tickets

import (
	"context"
	"crypto/rand"
	"encoding/base64"
	"errors"
	"fmt"
	"time"

	"github.com/redis/go-redis/v9"
)

var ErrInvalid = errors.New("invalid or expired ticket")

const prefix = "tickets:"

type Store struct {
	c   *redis.Client
	ttl time.Duration
}

func New(url string, ttl time.Duration) (*Store, error) {
	opt, err := redis.ParseURL(url)
	if err != nil {
		return nil, fmt.Errorf("parse REDIS_URL: %w", err)
	}
	return &Store{c: redis.NewClient(opt), ttl: ttl}, nil
}

func (s *Store) Close() error                   { return s.c.Close() }
func (s *Store) Ping(ctx context.Context) error { return s.c.Ping(ctx).Err() }

// Issue stores value under a new random ticket (256 bits) for the TTL.
func (s *Store) Issue(ctx context.Context, value string) (string, error) {
	buf := make([]byte, 32)
	if _, err := rand.Read(buf); err != nil {
		return "", err
	}
	ticket := base64.RawURLEncoding.EncodeToString(buf)
	if err := s.c.Set(ctx, prefix+ticket, value, s.ttl).Err(); err != nil {
		return "", err
	}
	return ticket, nil
}

// Redeem returns the ticket's value and deletes it atomically (GETDEL), so a
// ticket works exactly once.
func (s *Store) Redeem(ctx context.Context, ticket string) (string, error) {
	v, err := s.c.GetDel(ctx, prefix+ticket).Result()
	if errors.Is(err, redis.Nil) {
		return "", ErrInvalid
	}
	return v, err
}
```

```go
// Package auth issues and checks access tokens, refresh tokens and password
// hashes. Password hashes are compatible with the Node API's bcrypt hashes.
package auth

import (
	"errors"
	"time"

	"github.com/golang-jwt/jwt/v5"
)

// AccessTTL is how long an access token is valid; the UI refreshes it.
const AccessTTL = 15 * time.Minute

type Signer struct {
	secret []byte
	now    func() time.Time
}

func NewSigner(secret string) *Signer { return &Signer{secret: []byte(secret), now: time.Now} }

// Sign returns an HS256 token with claims {sub, iat, exp}.
func (s *Signer) Sign(userID string) (string, error) {
	now := s.now()
	claims := jwt.MapClaims{"sub": userID, "iat": now.Unix(), "exp": now.Add(AccessTTL).Unix()}
	return jwt.NewWithClaims(jwt.SigningMethodHS256, claims).SignedString(s.secret)
}

// Verify checks signature and expiry (exp is required) and returns sub.
func (s *Signer) Verify(token string) (string, error) {
	if token == "" {
		return "", errors.New("missing token")
	}
	parsed, err := jwt.Parse(token, func(*jwt.Token) (any, error) { return s.secret, nil },
		jwt.WithValidMethods([]string{jwt.SigningMethodHS256.Alg()}),
		jwt.WithExpirationRequired(),
		jwt.WithTimeFunc(s.now))
	if err != nil {
		return "", err
	}
	sub, err := parsed.Claims.GetSubject()
	if err != nil || sub == "" {
		return "", errors.New("token has no sub claim")
	}
	return sub, nil
}
```

```go
package auth

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/hex"
	"time"
)

// RefreshTTL is how long a refresh token is valid. Each use replaces it.
const RefreshTTL = 30 * 24 * time.Hour

// NewRefreshToken returns a random token for the client and the hash to store.
func NewRefreshToken() (plain, hash string, err error) {
	buf := make([]byte, 32)
	if _, err := rand.Read(buf); err != nil {
		return "", "", err
	}
	plain = base64.RawURLEncoding.EncodeToString(buf)
	return plain, HashRefreshToken(plain), nil
}

// HashRefreshToken is SHA-256 hex. Tokens are high-entropy random values, so
// a fast hash is enough; the database never sees the plain token.
func HashRefreshToken(plain string) string {
	sum := sha256.Sum256([]byte(plain))
	return hex.EncodeToString(sum[:])
}
```

```go
import (
```

```go
// API holds HTTP metrics. A nil *API is valid and records nothing.
type API struct {
	reg      *prometheus.Registry
	requests *prometheus.CounterVec
	duration *prometheus.HistogramVec
}

func NewAPI() *API {
	reg := prometheus.NewRegistry()
	m := &API{
		reg: reg,
		requests: prometheus.NewCounterVec(prometheus.CounterOpts{
			Name: "cp_http_requests_total", Help: "HTTP requests by route pattern, method and status.",
		}, []string{"route", "method", "status"}),
		duration: prometheus.NewHistogramVec(prometheus.HistogramOpts{
			Name: "cp_http_request_duration_seconds", Help: "HTTP request duration by route pattern.",
			Buckets: prometheus.DefBuckets,
		}, []string{"route"}),
	}
	reg.MustRegister(m.requests, m.duration,
		collectors.NewGoCollector(), collectors.NewProcessCollector(collectors.ProcessCollectorOpts{}))
	return m
}

// Observe records one request. route is the mux pattern (never the raw path,
// which would explode label cardinality with IDs).
func (m *API) Observe(route, method string, status int, d time.Duration) {
	if m == nil {
		return
	}
	m.requests.WithLabelValues(route, method, strconv.Itoa(status)).Inc()
	m.duration.WithLabelValues(route).Observe(d.Seconds())
}

func (m *API) Handler() http.Handler { return promhttp.HandlerFor(m.reg, promhttp.HandlerOpts{}) }
```

```go
package api

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"net/http"
	"regexp"
	"slices"
	"strings"
	"time"
)

type userKey struct{}
type requestIDKey struct{}

// userID returns the authenticated user set by private.
func userID(r *http.Request) string {
	id, _ := r.Context().Value(userKey{}).(string)
	return id
}

func requestID(r *http.Request) string {
	id, _ := r.Context().Value(requestIDKey{}).(string)
	return id
}

// private requires a valid Bearer access token.
func (s *Server) private(next http.HandlerFunc) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		token, _ := strings.CutPrefix(r.Header.Get("Authorization"), "Bearer ")
		uid, err := s.Signer.Verify(token)
		if err != nil {
			writeError(w, http.StatusUnauthorized, "unauthorized")
			return
		}
		next(w, r.WithContext(context.WithValue(r.Context(), userKey{}, uid)))
	})
}

var safeRequestID = regexp.MustCompile(`^[A-Za-z0-9-]{1,64}$`)

// withRequestID keeps a well-formed incoming X-Request-ID or generates one,
// echoes it in the response and stores it in the context for logs.
func withRequestID(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		id := r.Header.Get("X-Request-ID")
		if !safeRequestID.MatchString(id) {
			buf := make([]byte, 8)
			_, _ = rand.Read(buf)
			id = hex.EncodeToString(buf)
		}
		w.Header().Set("X-Request-ID", id)
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), requestIDKey{}, id)))
	})
}

// withCORS allows only the configured origins.
// withCORS must stay an exact-match allow-list (config rejects "*"). Adding
// Access-Control-Allow-Credentials needs a security re-review (CTO, 2026-10-07).
func withCORS(origins []string, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")
		allowed := origin != "" && slices.Contains(origins, origin)
		h := w.Header()
		if allowed {
			h.Set("Access-Control-Allow-Origin", origin)
			h.Set("Access-Control-Expose-Headers", "X-Request-ID")
		}
		h.Add("Vary", "Origin")
		if r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != "" {
			if allowed {
				h.Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
				h.Set("Access-Control-Allow-Headers", "Authorization, Content-Type, X-Request-ID")
				h.Set("Access-Control-Max-Age", "600")
			}
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

type statusWriter struct {
	http.ResponseWriter
	status int
}

func (w *statusWriter) WriteHeader(code int) {
	w.status = code
	w.ResponseWriter.WriteHeader(code)
}

// Unwrap lets http.ResponseController reach Flush on the real writer (SSE).
func (w *statusWriter) Unwrap() http.ResponseWriter { return w.ResponseWriter }

// withRecover logs each request (path only, never the query), records
// metrics by route pattern, and turns panics into 500s.
func (s *Server) withRecover(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		sw := &statusWriter{ResponseWriter: w, status: http.StatusOK}
		start := time.Now()
		defer func() {
			if rec := recover(); rec != nil {
				s.Log.Error("panic", "method", r.Method, "path", r.URL.Path, "requestId", requestID(r), "panic", rec)
				writeError(sw, http.StatusInternalServerError, "internal_error")
			}
			route := r.Pattern // set by ServeMux on this request
			if route == "" {
				route = "unmatched"
			}
			s.Metrics.Observe(route, r.Method, sw.status, time.Since(start))
			s.Log.Info("request", "method", r.Method, "path", r.URL.Path, "status", sw.status,
				"ms", time.Since(start).Milliseconds(), "requestId", requestID(r))
		}()
		next.ServeHTTP(sw, r)
	})
}
```

```go
package api

import (
	"net"
	"net/http"
	"sync"
	"time"

	"golang.org/x/time/rate"
)

// ipLimiter allows `burst` requests per client IP, refilling one every 6s
// (10/min). It is in memory: fine for one API instance (spec §12).
type ipLimiter struct {
	mu      sync.Mutex
	burst   int
	clients map[string]*client
	pruned  time.Time
}

type client struct {
	lim  *rate.Limiter
	seen time.Time
}

func newIPLimiter(burst int) *ipLimiter {
	return &ipLimiter{burst: burst, clients: map[string]*client{}, pruned: time.Now()}
}

func (l *ipLimiter) allow(ip string) bool {
	l.mu.Lock()
	defer l.mu.Unlock()
	now := time.Now()
	if now.Sub(l.pruned) > time.Minute {
		for k, c := range l.clients {
			if now.Sub(c.seen) > 10*time.Minute {
				delete(l.clients, k)
			}
		}
		l.pruned = now
	}
	c, ok := l.clients[ip]
	if !ok {
		c = &client{lim: rate.NewLimiter(rate.Every(6*time.Second), l.burst)}
		l.clients[ip] = c
	}
	c.seen = now
	return c.lim.Allow()
}

// wrap rejects requests over the limit with 429. The client IP is the TCP
// peer; X-Forwarded-For is not trusted.
func (l *ipLimiter) wrap(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ip, _, err := net.SplitHostPort(r.RemoteAddr)
		if err != nil {
			ip = r.RemoteAddr
		}
		if !l.allow(ip) {
			w.Header().Set("Retry-After", "60")
			writeError(w, http.StatusTooManyRequests, "too_many_requests")
			return
		}
		next.ServeHTTP(w, r)
	})
}
```

```go
package api

import (
	"context"
	"errors"
	"net/http"
	"net/mail"

	"infrapilot/internal/auth"
	"infrapilot/internal/store"
)

type authUser struct {
	ID    string  `json:"id"`
	Email string  `json:"email"`
	Name  *string `json:"name"`
}

type authResponse struct {
	Token        string   `json:"token"`
	RefreshToken string   `json:"refreshToken"`
	User         authUser `json:"user"`
}

func validEmail(s string) bool {
	a, err := mail.ParseAddress(s)
	return err == nil && a.Address == s
}

// issueTokens returns a new access token and a new stored refresh token.
func (s *Server) issueTokens(ctx context.Context, userID string) (string, string, error) {
	access, err := s.Signer.Sign(userID)
	if err != nil {
		return "", "", err
	}
	refresh, hash, err := auth.NewRefreshToken()
	if err != nil {
		return "", "", err
	}
	if err := s.Store.CreateRefreshToken(ctx, userID, hash, auth.RefreshTTL); err != nil {
		return "", "", err
	}
	return access, refresh, nil
}

func (s *Server) writeAuth(w http.ResponseWriter, r *http.Request, u store.User) {
	access, refresh, err := s.issueTokens(r.Context(), u.ID)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, authResponse{Token: access, RefreshToken: refresh,
		User: authUser{ID: u.ID, Email: u.Email, Name: u.Name}})
}

func (s *Server) register(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Email    string  `json:"email"`
		Password string  `json:"password"`
		Name     *string `json:"name"`
	}
	if err := decodeJSON(r, &body); err != nil || !validEmail(body.Email) || len(body.Password) < 8 || len(body.Password) > 72 {
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	}
	hash, err := auth.HashPassword(body.Password)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	u, err := s.Store.CreateUser(r.Context(), body.Email, hash, body.Name)
	if errors.Is(err, store.ErrConflict) {
		writeError(w, http.StatusConflict, "email_already_exists")
		return
	}
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	s.writeAuth(w, r, u)
}

func (s *Server) login(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Email    string `json:"email"`
		Password string `json:"password"`
	}
	if err := decodeJSON(r, &body); err != nil || !validEmail(body.Email) {
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	}
	u, err := s.Store.GetUserByEmail(r.Context(), body.Email)
	if errors.Is(err, store.ErrNotFound) || (err == nil && !auth.CheckPassword(u.Password, body.Password)) {
		writeError(w, http.StatusUnauthorized, "invalid_credentials")
		return
	}
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	s.writeAuth(w, r, u)
}

// refresh exchanges a refresh token for a new access + refresh token pair.
// Each refresh token works once; replaying one revokes all of the user's tokens.
func (s *Server) refresh(w http.ResponseWriter, r *http.Request) {
	var body struct {
		RefreshToken string `json:"refreshToken"`
	}
	if err := decodeJSON(r, &body); err != nil || body.RefreshToken == "" {
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	}
	userID, err := s.Store.ConsumeRefreshToken(r.Context(), auth.HashRefreshToken(body.RefreshToken))
	if errors.Is(err, store.ErrNotFound) || errors.Is(err, store.ErrTokenReused) {
		if errors.Is(err, store.ErrTokenReused) {
			s.Log.Warn("refresh token reused; revoked all sessions", "requestId", requestID(r))
		}
		writeError(w, http.StatusUnauthorized, "invalid_refresh_token")
		return
	}
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	access, refresh, err := s.issueTokens(r.Context(), userID)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]string{"token": access, "refreshToken": refresh})
}

func (s *Server) logout(w http.ResponseWriter, r *http.Request) {
	var body struct {
		RefreshToken string `json:"refreshToken"`
	}
	if err := decodeJSON(r, &body); err != nil {
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	}
	if body.RefreshToken != "" {
		if err := s.Store.RevokeRefreshToken(r.Context(), auth.HashRefreshToken(body.RefreshToken)); err != nil {
			s.internalError(w, r, err)
			return
		}
	}
	writeJSON(w, http.StatusOK, map[string]bool{"ok": true})
}

// devLogin finds or creates dev@local. Registered only when APP_ENV=development.
func (s *Server) devLogin(w http.ResponseWriter, r *http.Request) {
	u, err := s.Store.UpsertDevUser(r.Context())
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	access, refresh, err := s.issueTokens(r.Context(), u.ID)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]string{"token": access, "refreshToken": refresh, "userId": u.ID})
}
```

```go
func validURL(s string) bool {
	u, err := url.Parse(s)
	return err == nil && u.Scheme != "" && (u.Host != "" || u.Opaque != "")
}

// ownedService writes 404 and returns false unless the caller owns the service.
func (s *Server) ownedService(w http.ResponseWriter, r *http.Request, id string) (store.Service, bool) {
	svc, err := s.Store.GetServiceForOwner(r.Context(), id, userID(r))
	if errors.Is(err, store.ErrNotFound) {
		writeError(w, http.StatusNotFound, "service_not_found")
		return svc, false
	}
	if err != nil {
		s.internalError(w, r, err)
		return svc, false
	}
	return svc, true
}

func (s *Server) createService(w http.ResponseWriter, r *http.Request) {
	var body struct {
		ProjectID   string  `json:"projectId"`
		Name        string  `json:"name"`
		ServiceType *string `json:"serviceType"`
		RepoURL     *string `json:"repoUrl"`
		Branch      *string `json:"branch"`
	}
	if err := decodeJSON(r, &body); err != nil {
		writeError(w, http.StatusBadRequest, "invalid_json")
		return
	}
	typ := store.TypeDocker
	if body.ServiceType != nil {
		typ = *body.ServiceType
	}
	branch := "main"
	if body.Branch != nil {
		branch = *body.Branch
	}
	switch {
	case body.ProjectID == "" || utf8.RuneCountInString(body.Name) < 2 || !store.ValidServiceType(typ) || branch == "":
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	case body.RepoURL != nil && !validURL(*body.RepoURL):
		writeError(w, http.StatusBadRequest, "invalid_repo_url")
		return
	case store.NeedsRepo(typ) && body.RepoURL == nil:
		writeError(w, http.StatusBadRequest, "repoUrl required for docker/nodejs/nextjs/react services")
		return
	}
	if !s.ownsProject(w, r, body.ProjectID) {
		return
	}
	svc, err := s.Store.CreateService(r.Context(), store.NewService{

// Managed services skip the build (the workflow decides; the event type is
// kept for compatibility with Node-era rows).
func (s *Server) deployService(w http.ResponseWriter, r *http.Request) {

		CommitSha *string `json:"commitSha"`
	}
	if err := decodeJSON(r, &body); err != nil || (body.CommitSha != nil && utf8.RuneCountInString(*body.CommitSha) < 4) {
		writeError(w, http.StatusBadRequest, "invalid_request")
		return
	}
	svc, ok := s.ownedService(w, r, r.PathValue("id"))
	if !ok {
		return
	}
	managed := store.IsManaged(svc.ServiceType)
	sha, event := "main", store.EventBuildRequested
	if managed {
		sha, event = "latest", store.EventDeployRequested
	}
	if body.CommitSha != nil {
		sha = *body.CommitSha
	}

	ctx := r.Context()
	var dep store.Deployment
	err := s.Store.InTx(ctx, func(tx *store.Store) error {
		var err error
		if dep, err = tx.CreateDeployment(ctx, svc.ID, sha); err != nil {
			return err
		}
		_, err = tx.CreateOutboxEvent(ctx, event, map[string]string{"deploymentId": dep.ID})
		return err
	})
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]string{"deploymentId": dep.ID, "status": dep.Status})
}

// deleteService hides the service at once and queues TeardownWorkflow to
```

`internal/api/layout.go`:
```go
package api

const (
	nodeWidth  = 180.0
	nodeHeight = 80.0
	nodeSep    = 150.0 // vertical gap between nodes in one rank
	rankSep    = 200.0 // horizontal gap between ranks
)

type position struct {
	ID        string  `json:"id"`
	PositionX float64 `json:"positionX"`
	PositionY float64 `json:"positionY"`
}

// layout places nodes left to right. A node's rank is one more than its
// furthest predecessor's (longest path); nodes in a rank stack top to bottom
// in input order. Ranks are capped below len(ids), so cycles terminate. Edges
// to unknown nodes and self-loops are ignored.
func layout(ids []string, edges [][2]string) []position {
	known := make(map[string]bool, len(ids))
	for _, id := range ids {
		known[id] = true
	}
	rank := map[string]int{}
	for range ids {
		changed := false
		for _, e := range edges {
			from, to := e[0], e[1]
			if !known[from] || !known[to] || from == to {
				continue
			}
			if next := rank[from] + 1; next > rank[to] && next < len(ids) {
				rank[to] = next
				changed = true
			}
		}
		if !changed {
			break
		}
	}

	slot := map[int]int{}
	out := make([]position, 0, len(ids))
	for _, id := range ids {
		r := rank[id]
		out = append(out, position{
			ID:        id,
			PositionX: float64(r) * (nodeWidth + rankSep),
			PositionY: float64(slot[r]) * (nodeHeight + nodeSep),
		})
		slot[r]++
	}
	return out
}
```

```go
func parseLogQuery(deploymentID string, v url.Values) (logs.Query, error) {
	q := logs.Query{DeploymentID: deploymentID, Limit: 100, Search: v.Get("search")}
	if l, err := strconv.Atoi(v.Get("limit")); err == nil && l > 0 {
		q.Limit = min(l, 1000)
	}
	for name, dst := range map[string]**time.Time{"after": &q.Start, "before": &q.End} {
		raw := v.Get(name)
		if raw == "" {
			continue
		}
		t, err := time.Parse(time.RFC3339Nano, raw)
		if err != nil {
			return q, fmt.Errorf("invalid %s: expected an ISO-8601 timestamp", name)
		}
		*dst = &t
	}
	if src := v.Get("sources"); src != "" {
		q.Sources = strings.Split(src, ",")
	}
	return q, nil
}

// ownedDeployment writes 404 and returns false unless userID owns the deployment.
func (s *Server) ownedDeployment(w http.ResponseWriter, r *http.Request, id, uid string) bool {
	_, err := s.Store.GetDeploymentForOwner(r.Context(), id, uid)
	if errors.Is(err, store.ErrNotFound) {
		writeError(w, http.StatusNotFound, "deployment not found")
		return false
	}
	if err != nil {
		s.internalError(w, r, err)
		return false
	}
	return true
}

func (s *Server) getLogs(w http.ResponseWriter, r *http.Request) {
	id := r.PathValue("deploymentId")
	if !s.ownedDeployment(w, r, id, userID(r)) {
		return
	}
	q, err := parseLogQuery(id, r.URL.Query())
	if err != nil {
		writeError(w, http.StatusBadRequest, err.Error())
		return
	}
	res, err := s.Logs.Query(r.Context(), q)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	if res.Logs == nil {
		res.Logs = []logs.Entry{}
	}
	meta := logsMeta{Source: res.Source, Count: len(res.Logs), TotalCount: res.TotalCount}
	if res.TotalCount != nil && *res.TotalCount > 0 {
		meta.HasMore = len(res.Logs) < *res.TotalCount
	}
	writeJSON(w, http.StatusOK, struct {
		Logs []logs.Entry `json:"logs"`
		Meta logsMeta     `json:"meta"`
	}{res.Logs, meta})
}

	if err != nil {
		s.internalError(w, r, err)
		return
	}
```

```go
package api

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"
	"time"

	"infrapilot/internal/logs"
)

const sseHeartbeat = 15 * time.Second

type streamMeta struct {
	Source     string `json:"source"`
	TotalCount *int   `json:"totalCount,omitempty"`
}

// streamLogs authenticates with a one-time ticket (EventSource cannot send
// headers), then sends one "history" event ({logs, meta}), a "log" event per
// new line, and a comment every 15s so proxies keep the connection open.
func (s *Server) streamLogs(w http.ResponseWriter, r *http.Request) {
	id := r.PathValue("deploymentId")
	value, err := s.Tickets.Redeem(r.Context(), r.URL.Query().Get("ticket"))
	uid, ticketDep, _ := strings.Cut(value, ":")
	if err != nil || ticketDep != id || uid == "" {
		writeError(w, http.StatusUnauthorized, "invalid_ticket")
		return
	}
	if !s.ownedDeployment(w, r, id, uid) { // the service may have been deleted since
		return
	}
	ctx := r.Context()
	// Subscribe before reading history so lines written in between are not
	// lost; the UI drops duplicates (same timestamp and message).
	live, stop, err := s.Live.Subscribe(ctx, id)
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	defer stop()
	hist, err := s.Logs.Query(ctx, logs.Query{DeploymentID: id, Limit: 100})
	if err != nil {
		s.internalError(w, r, err)
		return
	}
	if hist.Logs == nil {
		hist.Logs = []logs.Entry{}
	}

	h := w.Header()
	h.Set("Content-Type", "text/event-stream")
	h.Set("Cache-Control", "no-cache")
	h.Set("Connection", "keep-alive")
	h.Set("X-Accel-Buffering", "no")
	w.WriteHeader(http.StatusOK)
	rc := http.NewResponseController(w)
	send := func(event string, v any) error {
		if err := writeEvent(w, event, v); err != nil {
			return err
		}
		return rc.Flush()
	}

	history := map[string]any{"logs": hist.Logs, "meta": streamMeta{Source: hist.Source, TotalCount: hist.TotalCount}}
	if send("history", history) != nil {
		return
	}
	ping := time.NewTicker(sseHeartbeat)
	defer ping.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case e, ok := <-live:
			if !ok || send("log", e) != nil {
				return
			}
		case <-ping.C:
			if _, err := io.WriteString(w, ": ping\n\n"); err != nil || rc.Flush() != nil {
				return
			}
		}
	}
}

// writeEvent writes one SSE event. json.Marshal never emits raw newlines, so
// the data fits on one line.
func writeEvent(w io.Writer, event string, v any) error {
	b, err := json.Marshal(v)
	if err != nil {
		return err
	}
	_, err = fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, b)
	return err
}
```

```go
func TeardownWorkflow(ctx workflow.Context, in TeardownInput) error {

func (a *Activities) TeardownService(ctx context.Context, in TeardownInput) error {
	live, err := a.Store.ListLiveDeployments(ctx, in.ServiceID, "")

	}
	if err := a.Router.RemoveRoute(ctx, in.ServiceID); err != nil {
		return err
	}
	if in.PurgeData {
		return a.Docker.RemoveVolume(ctx, VolumeName(in.ServiceID))
	}
	return nil
```

```ts
export const API_BASE: string = import.meta.env.VITE_API_BASE ?? "http://localhost:8080";

const ACCESS_KEY = "cp_token";
const REFRESH_KEY = "cp_refresh";

export function getToken(): string {
  return localStorage.getItem(ACCESS_KEY) ?? "";
}

export function setSession(token: string, refreshToken: string): void {
  localStorage.setItem(ACCESS_KEY, token);
  localStorage.setItem(REFRESH_KEY, refreshToken);
}

export function clearSession(): void {
  localStorage.removeItem(ACCESS_KEY);
  localStorage.removeItem(REFRESH_KEY);
}

// One refresh at a time: parallel 401s wait for the same request.
let refreshing: Promise<boolean> | null = null;

function refreshSession(): Promise<boolean> {
  const refreshToken = localStorage.getItem(REFRESH_KEY);
  if (!refreshToken) return Promise.resolve(false);
  refreshing ??= (async () => {
    try {
      const res = await fetch(`${API_BASE}/auth/refresh`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ refreshToken }),
      });
      if (!res.ok) return false;
      const data = (await res.json()) as { token: string; refreshToken: string };
      setSession(data.token, data.refreshToken);
      return true;
    } catch {
      return false;
    } finally {
      refreshing = null;
    }
  })();
  return refreshing;
}

export async function apiFetch<T>(path: string, opts: RequestInit = {}): Promise<T> {
  const send = () => {
    const headers = new Headers(opts.headers);
    const token = getToken();
    if (token) headers.set("Authorization", `Bearer ${token}`);
    if (opts.body != null && !headers.has("Content-Type")) headers.set("Content-Type", "application/json");
    return fetch(`${API_BASE}${path}`, { ...opts, headers });
  };

  const isAuthCall = path.startsWith("/auth/");
  let res = await send();
  if (res.status === 401 && !isAuthCall && getToken()) {
    if (await refreshSession()) {
      res = await send();
    } else {
      clearSession();
      window.location.assign("/login");
    }
  }
  if (!res.ok) {
    const text = await res.text();
    throw new Error(`${res.status}: ${text}`);
  }
  return (await res.json()) as T;
}

// logout revokes the refresh token on the server and clears local tokens.
export async function logout(): Promise<void> {
  const refreshToken = localStorage.getItem(REFRESH_KEY);
  clearSession();
  if (!refreshToken) return;
  try {
    await fetch(`${API_BASE}/auth/logout`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ refreshToken }),
    });
  } catch {
    // Offline: the token still expires on its own.
  }
}
```

```ts
import { useState, useEffect } from "react";
import { API_BASE, apiFetch, getToken } from "../lib/api";

export interface LogEntry {
  deploymentId: string;
  source: "stdout" | "stderr" | "system";
  message: string;
  timestamp: string;
}

interface LogMeta {
  source: "redis" | "clickhouse";
  totalCount?: number;
}

interface UseLogsOptions {
  deploymentId: string;
}

const MAX_RECONNECTS = 5;

// Streams a deployment's logs over Server-Sent Events. Each connection uses a
// fresh one-time ticket (EventSource cannot send the Authorization header).
// The server sends one "history" event, then a "log" event per line.
export function useLogs({ deploymentId }: UseLogsOptions) {
  const [logs, setLogs] = useState<LogEntry[]>([]);
  const [meta, setMeta] = useState<LogMeta | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (!getToken() || !deploymentId) return;

    let source: EventSource | null = null;
    let retryTimer: ReturnType<typeof setTimeout> | undefined;
    let attempts = 0;
    let closed = false;

    setLogs([]);
    setMeta(null);
    setError(null);

    const path = `/logs/deployments/${encodeURIComponent(deploymentId)}`;

    function scheduleReconnect() {
      if (closed) return;
      attempts += 1;
      if (attempts > MAX_RECONNECTS) {
        setError("Connection lost. Reopen the logs to retry.");
        return;
      }
      retryTimer = setTimeout(connect, Math.min(1000 * 2 ** attempts, 15000));
    }

    async function connect() {
      try {
        const { ticket } = await apiFetch<{ ticket: string }>(`${path}/stream-ticket`, { method: "POST" });
        if (closed) return;
        source = new EventSource(`${API_BASE}${path}/stream?ticket=${encodeURIComponent(ticket)}`);
        source.onopen = () => {
          attempts = 0;
          setIsConnected(true);
          setError(null);
        };
        source.onerror = () => {
          source?.close();
          setIsConnected(false);
          scheduleReconnect();
        };
        source.addEventListener("history", (ev) => {
          const data = JSON.parse((ev as MessageEvent<string>).data) as { logs: LogEntry[]; meta: LogMeta };
          setLogs(data.logs);
          setMeta(data.meta);
        });
        source.addEventListener("log", (ev) => {
          const log = JSON.parse((ev as MessageEvent<string>).data) as LogEntry;
          setLogs((prev) =>
            prev.some((l) => l.timestamp === log.timestamp && l.message === log.message) ? prev : [...prev, log]
          );
        });
      } catch {
        setIsConnected(false);
        setError("Connection error");
        scheduleReconnect();
      }
    }

    void connect();
    return () => {
      closed = true;
      clearTimeout(retryTimer);
      source?.close();
      setIsConnected(false);
    };
  }, [deploymentId]);

  const clearLogs = () => {
    setLogs([]);
  };

  return { logs, meta, isConnected, error, clearLogs };
}
```

```bash
# Compares read-only responses of the Node API and the Go API against the same

TOKEN=$(curl -sf -X POST "$GO/dev/login" | jq -r .token)

status=0
check() { # path, jq filter
  if diff <(curl -sf "${auth[@]}" "$NODE$1" | jq -S "$2") \
          <(curl -sf "${auth[@]}" "$GO$1"   | jq -S "$2") > /tmp/compare-api.diff; then

check "/services?projectId=$PROJECT" "$NORMALIZE"

check "/projects/$PROJECT/canvas" "$NORMALIZE"
```
